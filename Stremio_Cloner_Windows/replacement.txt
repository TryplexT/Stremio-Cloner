              _log("[PROGRESS] Patching files... 100%");
              _log(
                "[SUCCESS] Patched $patchedCount images and removed $killedCount XMLs in res/.",
              );

              _log("[PROGRESS] Patching manifest and strings...");
              final manifestFile = File(p.join(_decompiledDirPath!, 'AndroidManifest.xml'));
              if (await manifestFile.exists()) {
                String manifestContent = await manifestFile.readAsString();
                manifestContent = manifestContent.replaceAll('package="com.stremio.one"', 'package="$newPackage"');
                manifestContent = manifestContent.replaceAll('com.stremio.one', newPackage);
                manifestContent = manifestContent.replaceAll('android:extractNativeLibs="false"', 'android:extractNativeLibs="true"');
                await manifestFile.writeAsString(manifestContent);
              }

              final apktoolFile = File(p.join(_decompiledDirPath!, 'apktool.yml'));
              if (await apktoolFile.exists()) {
                String apktoolContent = await apktoolFile.readAsString();
                apktoolContent = apktoolContent.replaceAll('renameManifestPackage: com.stremio.one', 'renameManifestPackage: $newPackage');
                await apktoolFile.writeAsString(apktoolContent);
              }

              final stringsFile = File(p.join(_decompiledDirPath!, 'res', 'values', 'strings.xml'));
              if (await stringsFile.exists()) {
                String stringsContent = await stringsFile.readAsString();
                final appNameRegex = RegExp(r'<string name="app_name">.*?</string>');
                if (stringsContent.contains(appNameRegex)) {
                  stringsContent = stringsContent.replaceFirst(appNameRegex, '<string name="app_name">$appName</string>');
                } else {
                  final regex2 = RegExp(r'<string name="app_name".*?>.*?</string>');
                  stringsContent = stringsContent.replaceFirst(regex2, '<string name="app_name">$appName</string>');
                }
                await stringsFile.writeAsString(stringsContent);
              }

              _log("[PROGRESS] Rebuilding APK with apktool...");
              final unalignedApkPath = p.join(outputDir.path, "$appName-unaligned.apk");
              final buildResult = await Process.run('java', [
                '-jar',
                apktoolJar,
                'b',
                _decompiledDirPath!,
                '-o',
                unalignedApkPath
              ]);

              if (buildResult.exitCode == 0) {
                 _log("[SUCCESS] APK Rebuilt.");
                 _log("[PROGRESS] Signing APK with uber-apk-signer...");
                 
                 final signResult = await Process.run('java', [
                   '-jar',
                   _signerJar!,
                   '-a',
                   unalignedApkPath,
                   '--overwrite'
                 ]);
                 
                 if (signResult.exitCode == 0) {
                    _log("[SUCCESS] APK Signed.");
                    final signedFile = File(unalignedApkPath);
                    if (await signedFile.exists()) {
                        await signedFile.rename(finalApkPath);
                    }
                 } else {
                    _log("[ERROR] Signer failed: ${signResult.stderr}");
                 }
              } else {
                 _log("[ERROR] Apktool build failed: ${buildResult.stderr}");
              }
            } else {
              _log("[ERROR] res/ folder not found in decompiled directory.");
            }
